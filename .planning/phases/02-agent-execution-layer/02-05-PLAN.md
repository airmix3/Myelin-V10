**src/app/api/hire_requests/[id]/approve/route.ts** -- Per D-11, AGENT-06:

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';
import { sqlite } from '@/lib/db';
import { generateId } from '@/lib/id';
import { transitionTask } from '@/lib/state-machine';
import { eventBus } from '@/lib/events';
import { logger } from '@/lib/logger';

const log = logger.child({ module: 'hire-approve' });

export async function POST(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;

  // Find hire request
  const hireRequest = await prisma.hireRequest.findUnique({ where: { id } });
  if (!hireRequest) {
    return NextResponse.json({ error: 'Hire request not found' }, { status: 404 });
  }
  if (hireRequest.status !== 'pending') {
    return NextResponse.json({ error: `Hire request already ${hireRequest.status}` }, { status: 400 });
  }

  // Create temp employee record
  const employeeId = generateId('emp');
  const agentId = `temp_${hireRequest.employeeName.toLowerCase().replace(/\s+/g, '_')}_${Date.now()}`;
  await prisma.employee.create({
    data: {
      id: employeeId,
      name: hireRequest.employeeName,
      role: 'temp',
      department: /* get from task */ '',
      agentId,
      status: 'active',
      budgetLimit: 5.0,  // Lower budget for temps
      budgetSpent: 0.0,
    },
  });

  // Look up task for department
  const task = await prisma.task.findUnique({ where: { id: hireRequest.taskId } });
  if (task) {
    // Update employee department from task
    await prisma.employee.update({
      where: { id: employeeId },
      data: { department: task.department },
    });
  }

  // Update hire request status
  await prisma.hireRequest.update({
    where: { id },
    data: { status: 'approved', resolvedAt: new Date() },
  });

  // Store the subagent definition in task metadata
  // Per AGENT-06: employee sandbox has restricted tools only
  const subagentDef = {
    [agentId]: {
      description: hireRequest.employeeRole,
      prompt: `You are ${hireRequest.employeeName}, a ${hireRequest.employeeRole} working under the ${task?.department ?? 'unknown'} department.\nYour task context: ${task?.title ?? ''}\n${task?.description ?? ''}\nYou have limited tool access. Focus on producing deliverables.`,
      tools: [
        'Read', 'Edit', 'Write', 'Bash', 'Glob', 'Grep',
        'mcp__myelin__read_memory', 'mcp__myelin__write_memory',
        'mcp__myelin__read_knowledge', 'mcp__myelin__search_knowledge',
        'mcp__myelin__promote_to_deliverable', 'mcp__myelin__propose_skill',
        'mcp__myelin__submit_for_review',
      ],
    },
  };

  // Store subagent definition in task metadata for the dept head's next invocation
  const existingMetadata = task?.metadata ? JSON.parse(task.metadata) : {};
  const updatedMetadata = {
    ...existingMetadata,
    inputType: undefined,  // clear the hire_approval input type
    hireApproved: true,
    hireRequestId: id,
    employeeId,
    employeeAgentId: agentId,
    subagentDefinition: subagentDef,
  };
  sqlite.prepare('UPDATE tasks SET metadata = ? WHERE id = ?')
    .run(JSON.stringify(updatedMetadata), hireRequest.taskId);

  // Transition task back to working
  transitionTask(hireRequest.taskId, 'input-required', 'working', {
    hireApproved: true,
    employeeId,
    employeeName: hireRequest.employeeName,
  });

  // Emit event
  eventBus.emit('hire:approved', {
    taskId: hireRequest.taskId,
    hireRequestId: id,
    employeeId,
    employeeName: hireRequest.employeeName,
    employeeRole: hireRequest.employeeRole,
  });

  log.info({ hireRequestId: id, employeeId, agentId }, 'Hire request approved');

  return NextResponse.json({
    success: true,
    employeeId,
    agentId,
    message: `${hireRequest.employeeName} hired as ${hireRequest.employeeRole}`,
  });
}
```

Note on Next.js 14 params: The `params` in Next.js 14 App Router is passed directly (not as a Promise). Use `const { id } = params;` if TypeScript complains about the Promise type. Check the existing Next.js version pattern.

export async function POST(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  const body = await request.json().catch(() => ({}));
  const reason = (body as { reason?: string }).reason ?? 'No reason provided';

  const hireRequest = await prisma.hireRequest.findUnique({ where: { id } });
  if (!hireRequest) {
    return NextResponse.json({ error: 'Hire request not found' }, { status: 404 });
  }
  if (hireRequest.status !== 'pending') {
    return NextResponse.json({ error: `Hire request already ${hireRequest.status}` }, { status: 400 });
  }

  // Update hire request
  await prisma.hireRequest.update({
    where: { id },
    data: { status: 'rejected', resolvedAt: new Date() },
  });

  // Update task metadata with rejection context
  const task = await prisma.task.findUnique({ where: { id: hireRequest.taskId } });
  const existingMetadata = task?.metadata ? JSON.parse(task.metadata) : {};
  const updatedMetadata = {
    ...existingMetadata,
    inputType: undefined,
    hireRejected: true,
    hireRequestId: id,
    rejectionReason: reason,
  };
  sqlite.prepare('UPDATE tasks SET metadata = ? WHERE id = ?')
    .run(JSON.stringify(updatedMetadata), hireRequest.taskId);

  // Transition back to working -- dept head continues without the hire
  transitionTask(hireRequest.taskId, 'input-required', 'working', {
    hireRejected: true,
    rejectionReason: reason,
  });

  eventBus.emit('hire:rejected', {
    taskId: hireRequest.taskId,
    hireRequestId: id,
    reason,
  });

  return NextResponse.json({
    success: true,
    message: `Hire request for ${hireRequest.employeeName} rejected: ${reason}`,
  });
}

```typescript
/**
 * JSON Schema objects for SDK outputFormat.
 * Per AGENT-04: SDK structured output for ALL turn routing -- no regex parsing.
 * Per research Pitfall 4: outputFormat takes raw JSON Schema, not zod.
 *
 * These schemas match the TypeScript interfaces in ./types.ts:
 *   ROUTING_SCHEMA -> RoutingResult
 *   AGENT_TURN_SCHEMA -> AgentTurnResult
 */

export const ROUTING_SCHEMA = {
  type: 'json_schema' as const,
  schema: {
    type: 'object',
    properties: {
      department: {
        type: 'string',
        enum: ['tech', 'marketing', 'operations'],
        description: 'Which department should handle this task',
      },
      confidence: {
        type: 'number',
        description: 'Confidence score 0-1 for the routing decision',
      },
      reasoning: {
        type: 'string',
        description: 'Brief explanation of why this department was chosen',
      },
      suggested_title: {
        type: 'string',
        description: 'A concise title for the task (under 80 characters)',
      },
    },
    required: ['department', 'confidence', 'reasoning', 'suggested_title'],
    additionalProperties: false,
  },
};

export const AGENT_TURN_SCHEMA = {
  type: 'json_schema' as const,
  schema: {
    type: 'object',
    properties: {
      turn_type: {
        type: 'string',
        enum: ['question', 'clarification', 'plan_ready', 'plan_update', 'done'],
        description: 'Type of this conversational turn',
      },
      message: {
        type: 'string',
        description: 'The agent message to display to the user',
      },
      plan_markdown: {
        type: 'string',
        description: 'Full plan in markdown format. Present when turn_type is plan_ready or plan_update.',
      },
    },
    required: ['turn_type', 'message'],
    additionalProperties: false,
  },
};
```

Export both constants. These are passed directly to `invokeAgent({ outputFormat: ROUTING_SCHEMA })` or `orchestrator.invoke({ outputFormat: AGENT_TURN_SCHEMA })` by the Tamir routing and planning conversation flows (built in Phase 3).
