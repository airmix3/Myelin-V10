Implementation of `invokeAgent(opts: InvokeAgentOptions): Promise<InvokeAgentResult>`:

1. Create ToolContext via `createToolContext({ taskId, agentId, department, deskDir, delivDir, manifestPath })`
2. Create MCP server: `const myelinServer = buildMyelinMcpServer(ctx)`
3. Create canUseTool: `const canUseTool = buildCanUseTool(opts.agentId, opts.department)`
4. Create a child logger: `const log = logger.child({ module: 'invoke-agent', taskId: opts.taskId, agentId: opts.agentId })`
5. Call `query()` with these options:
   - `prompt: opts.prompt`
   - `options.systemPrompt: { type: 'preset', preset: 'claude_code', append: opts.soulMd }`
   - `options.cwd: opts.deskDir`
   - `options.mcpServers: { myelin: myelinServer }`
   - `options.allowedTools: ['mcp__myelin__*']` -- auto-approve all myelin tools
   - `options.canUseTool: canUseTool`
   - `options.permissionMode: 'bypassPermissions'`
   - `options.allowDangerouslySkipPermissions: true`
   - `options.maxBudgetUsd: opts.maxBudgetUsd ?? 10`
   - `options.includePartialMessages: true`
   - `options.resume: opts.sessionId` (undefined on first call, session_id on resume)
   - `options.outputFormat: opts.outputFormat` (only if provided)
   - `options.agents: opts.agents` (only if provided)
   - `options.settingSources: ['project']` -- loads desk/CLAUDE.md
   - `options.env: { ...process.env as Record<string, string>, CLAUDE_CODE_USE_BEDROCK: '1', CLAUDE_CODE_STREAM_CLOSE_TIMEOUT: '120000' }`

6. Iterate the async generator `for await (const msg of q)`:
   - `msg.type === 'system' && msg.subtype === 'init'`: capture `msg.session_id`, log MCP server status, check for failed MCP connections
   - `msg.type === 'system' && msg.subtype === 'api_retry'`: log warning with attempt and retry_delay_ms
   - `msg.type === 'assistant'`: log to activity_log table with actionType='SDK_ASSISTANT', emit `eventBus.emit('task:buildlog', { taskId, event: { type: 'assistant', agentId: opts.agentId, content: msg } })`
   - `msg.type === 'stream_event'`: emit `eventBus.emit('task:buildlog', { taskId, event: msg })`
   - `msg.type === 'tool_progress'`: emit to buildlog
   - `msg.type === 'tool_use_summary'`: log to activity_log with actionType='SDK_TOOL_SUMMARY'
   - `msg.type === 'result'`: handle below

7. On result message:
   - If `msg.subtype === 'success'`: extract `total_cost_usd`, `result`, `structured_output`, `usage`, `modelUsage`
   - If `msg.subtype` starts with 'error': log error, throw with descriptive message
   - Log cost event to `cost_events` table: `sqlite.prepare('INSERT INTO cost_events (id, taskId, agentId, inputTokens, outputTokens, costUsd, model, createdAt) VALUES (?, ?, ?, ?, ?, ?, ?, datetime(\'now\'))').run(generateId('cost'), opts.taskId, opts.agentId, usage?.input_tokens ?? 0, usage?.output_tokens ?? 0, total_cost_usd, modelUsage?.model ?? 'unknown')`
   - Store session: `sqlite.prepare('UPDATE task_runs SET sessionId = ?, workspaceCwd = ? WHERE id = ?').run(sessionId, opts.deskDir, opts.runId)`
   - Return `{ sessionId, totalCostUsd: total_cost_usd, result, structuredOutput }`

8. After the loop (shouldn't reach here normally): throw `'Query ended without result message'`

9. Log activity entries using `sqlite.prepare('INSERT INTO activity_log ...').run(...)` with generateId('log').

```typescript
import { logger } from '@/lib/logger';
import { invokeAgent, type InvokeAgentOptions, type InvokeAgentResult } from '@/lib/invoke-agent';

interface AgentRegistryEntry {
  agentId: string;
  name: string;
  department: string;
  role: string;
  soulMd: string;
  avatarColor: string;
}

export class AgentOrchestrator {
  private agents = new Map<string, AgentRegistryEntry>();
  private log = logger.child({ module: 'orchestrator' });

  register(entry: AgentRegistryEntry): void {
    this.agents.set(entry.agentId, entry);
    this.log.info({ agentId: entry.agentId, name: entry.name }, 'Agent registered');
  }

  getAgent(agentId: string): AgentRegistryEntry | undefined {
    return this.agents.get(agentId);
  }

  getAllAgents(): AgentRegistryEntry[] {
    return Array.from(this.agents.values());
  }

  async invoke(opts: {
    taskId: string;
    runId: string;
    agentId: string;
    prompt: string;
    deskDir: string;
    delivDir: string;
    manifestPath: string;
    sessionId?: string;
    maxBudgetUsd?: number;
    outputFormat?: { type: 'json_schema'; schema: Record<string, unknown> };
    agents?: Record<string, unknown>;
  }): Promise<InvokeAgentResult> {
    const agent = this.agents.get(opts.agentId);
    if (!agent) throw new Error(`Unknown agent: ${opts.agentId}`);

    return invokeAgent({
      ...opts,
      department: agent.department,
      soulMd: agent.soulMd,
    });
  }
}
```

```typescript
async function executeRun(run: Record<string, unknown>): Promise<void> {
  const runId = run.id as string;
  const taskId = run.taskId as string;
  const employeeId = run.employeeId as string;

  const log = logger.child({ module: 'worker', runId, taskId, employeeId });
  log.info('Executing run');

  startHeartbeat(runId);

  // Emit worker claim event
  eventBus.emit('worker:claim', { runId, taskId, employeeId });

  // Log to activity_log
  sqlite.prepare(`
    INSERT INTO activity_log (id, taskId, agentId, actionType, description, createdAt)
    VALUES (?, ?, ?, 'WORKER_CLAIM', ?, datetime('now'))
  `).run(generateId('log'), taskId, employeeId, `Run ${runId} claimed by worker`);

  try {
    // Look up the task for context
    const task = sqlite.prepare('SELECT * FROM tasks WHERE id = ?').get(taskId) as Record<string, unknown> | undefined;
    if (!task) throw new Error(`Task ${taskId} not found`);

    // Look up the employee to get agentId
    const employee = sqlite.prepare('SELECT * FROM employees WHERE id = ?').get(employeeId) as Record<string, unknown> | undefined;
    if (!employee) throw new Error(`Employee ${employeeId} not found`);

    const agentId = employee.agentId as string;
    const department = task.department as string;

    // Get workspace paths from the task run or task
    const workspaceCwd = run.workspaceCwd as string | null;
    let deskDir: string;
    let delivDir: string;
    let manifestPath: string;

    if (workspaceCwd) {
      // Resuming -- use stored workspace path
      deskDir = workspaceCwd;
      const baseDir = path.dirname(deskDir);
      delivDir = path.join(baseDir, 'deliverables');
      manifestPath = path.join(delivDir, 'deliverable_manifest.json');
    } else {
      // First run -- workspace should already be created by the approval flow
      // Fall back to constructing from convention
      const baseDir = path.resolve(process.cwd(), 'data', 'workspaces', taskId);
      deskDir = path.join(baseDir, 'desk');
      delivDir = path.join(baseDir, 'deliverables');
      manifestPath = path.join(delivDir, 'deliverable_manifest.json');
    }

    // Build the prompt from task description + plan
    const prompt = [
      task.title as string,
      task.description ? `\n\n${task.description}` : '',
      task.planMarkdown ? `\n\n## Approved Plan\n\n${task.planMarkdown}` : '',
    ].join('');

    // Transition task to working if it's in submitted state
    const currentState = task.state as string;
    if (currentState === 'submitted') {
      transitionTask(taskId, 'submitted', 'working');
    }

    // Invoke the agent via orchestrator
    const result = await orchestrator.invoke({
      taskId,
      runId,
      agentId,
      prompt,
      deskDir,
      delivDir,
      manifestPath,
      sessionId: (run.sessionId as string) || undefined,
      maxBudgetUsd: (employee.budgetLimit as number) ?? 10,
    });

    log.info({ sessionId: result.sessionId, costUsd: result.totalCostUsd }, 'Agent invocation completed');

    // Update employee budget spent
    sqlite.prepare(
      'UPDATE employees SET budgetSpent = budgetSpent + ?, updatedAt = datetime(\'now\') WHERE id = ?'
    ).run(result.totalCostUsd, employeeId);

    // Mark run completed
    sqlite.prepare(`
      UPDATE task_runs
      SET status = 'completed', completedAt = datetime('now')
      WHERE id = ?
    `).run(runId);

    // Emit completion event
    eventBus.emit('agent:completed', { taskId, runId, agentId, costUsd: result.totalCostUsd });

    log.info({ runId, taskId, costUsd: result.totalCostUsd }, 'Run completed');
  } catch (err) {
    log.error({ err, runId, taskId }, 'Run execution failed');

    // Mark run failed
    sqlite.prepare(`
      UPDATE task_runs
      SET status = 'failed', failedAt = datetime('now'), failureReason = ?
      WHERE id = ?
    `).run(err instanceof Error ? err.message : 'unknown_error', runId);

    // Transition task to failed
    const task = sqlite.prepare('SELECT state FROM tasks WHERE id = ?').get(taskId) as { state: string } | undefined;
    if (task && task.state === 'working') {
      transitionTask(taskId, 'working', 'failed');
    }
  } finally {
    stopHeartbeat(runId);
  }
}
```
