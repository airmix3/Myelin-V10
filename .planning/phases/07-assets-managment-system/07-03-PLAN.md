```typescript
export interface Viewport {
  offsetX: number;  // world X at left edge of screen
  offsetY: number;  // world Y at top edge of screen
  scale: number;    // pixels per world unit
}

export const DEFAULT_VIEWPORT: Viewport = { offsetX: -200, offsetY: -200, scale: 1.0 };
export const MIN_SCALE = 0.5;
export const MAX_SCALE = 3.0;

export function worldToScreen(wx: number, wy: number, vp: Viewport): [number, number] {
  return [(wx - vp.offsetX) * vp.scale, (wy - vp.offsetY) * vp.scale];
}

export function screenToWorld(sx: number, sy: number, vp: Viewport): [number, number] {
  return [sx / vp.scale + vp.offsetX, sy / vp.scale + vp.offsetY];
}

/** Zoom centered on a screen point */
export function zoomAt(vp: Viewport, screenX: number, screenY: number, delta: number): Viewport {
  const factor = delta > 0 ? 0.9 : 1.1;
  const newScale = Math.max(MIN_SCALE, Math.min(MAX_SCALE, vp.scale * factor));
  const [wx, wy] = screenToWorld(screenX, screenY, vp);
  return {
    scale: newScale,
    offsetX: wx - screenX / newScale,
    offsetY: wy - screenY / newScale,
  };
}
```

```typescript
import type { AssetRenderData } from './buildings';
import { screenToWorld, type Viewport } from './viewport';

export function hitTest(
  screenX: number, screenY: number,
  assets: AssetRenderData[], vp: Viewport
): string | null {
  const [wx, wy] = screenToWorld(screenX, screenY, vp);
  // Iterate in reverse (last drawn = on top = highest priority)
  for (let i = assets.length - 1; i >= 0; i--) {
    const a = assets[i];
    // Hit box: building rect from (x - width/2, y - height) to (x + width/2, y)
    // Minimum 44px touch target per UI-SPEC
    const hitW = Math.max(a.width, 44) / vp.scale;
    const hitH = Math.max(a.height, 44) / vp.scale;
    if (wx >= a.x - hitW / 2 && wx <= a.x + hitW / 2 &&
        wy >= a.y - hitH && wy <= a.y) {
      return a.id;
    }
  }
  return null;
}

export function getTooltipText(asset: AssetRenderData): string {
  return `${asset.title} \u2014 ${asset.maturity}`;
}
```

**3. Create `src/app/assets/AssetCityCanvas.tsx`** (the main Canvas component):

Follow the NeuralHero.tsx pattern strictly:
- `'use client'` at top
- Props: `{ assets: Asset[], selectedAssetId: string | null, onAssetSelect: (id: string | null) => void }`
- Refs: `canvasRef = useRef<HTMLCanvasElement>(null)`, `vpRef = useRef<Viewport>(DEFAULT_VIEWPORT)`, `assetsRef = useRef<AssetRenderData[]>([])`, `animRef = useRef<number>(0)`, `hoveredRef = useRef<string | null>(null)`, `isDraggingRef = useRef(false)`, `dragStartRef = useRef<{x:number,y:number}|null>(null)`, `timeRef = useRef(0)`

**Data preparation (useEffect on assets change):**
- Parse each asset into AssetRenderData:
  - Parse `returnFactors` from JSON string to string array
  - Compute position using `assignPosition(asset.id, asset.category, [])` from districts.ts
  - Compute width (30) and height from `MATURITY_HEIGHT[asset.maturity]`
- Store in assetsRef.current

**Animation loop (useEffect on mount):**
- Similar to NeuralHero: `requestAnimationFrame` loop
- `timeRef.current += 0.016` per frame
- Dirty flag pattern: only redraw when `dirtyRef.current` is true (set true on viewport change, hover change, asset data change, or every 60th frame for pulse animations)
- Draw sequence:
  1. Clear canvas with `#1a1a2e` (var(--bg))
  2. For each category with at least one asset: `drawDistrictBoundary`, `drawDistrictLabel`
  3. Draw dependency lines: for each AssetDependency, draw dashed line between source and target buildings. Color `#2a2a4a`, opacity 0.3, `setLineDash([6, 3])`
  4. For each asset in assetsRef: call `drawBuilding(ctx, asset, vpRef.current)`. Then if asset has returnFactors, call `drawGlow`. Then if unhealthy, call `drawDegradation`. Then if recently active, call `drawActivityParticles`.
  5. Draw selection highlight: if `selectedAssetId` matches an asset, draw accent-colored ring (`#e94560`) around the building (2px stroke, 3px offset from building bounds)
  6. Draw hover highlight: if `hoveredRef.current` matches a different asset from selected, draw lighter ring

**ResizeObserver (useEffect on mount):**
- Observe canvas parent element
- On resize: set `canvas.width = parent.clientWidth`, `canvas.height = parent.clientHeight`, set dirtyRef true

**Mouse event handlers (on canvas element):**
- `onMouseDown`: Record drag start position. Set `isDraggingRef = true`.
- `onMouseMove`: If dragging, compute delta, update `vpRef.current.offsetX -= dx / vpRef.current.scale`, same for Y. Set dirty. Else: run hitTest, update `hoveredRef`, set cursor to 'pointer' or 'grab'. Set dirty if hovered changed.
- `onMouseUp`: If drag distance < 5px (was a click, not a pan): run hitTest. If hit, call `onAssetSelect(hitId)`. If no hit and panel is open, call `onAssetSelect(null)`. Reset drag state.
- `onWheel`: Call `zoomAt(vpRef.current, e.offsetX, e.offsetY, e.deltaY)`, update vpRef, set dirty. `e.preventDefault()`.
- `onDoubleClick`: Hit test. If hit, animate viewport to center on that building (smooth transition over 300ms using lerp in the animation loop).

**Canvas tooltip:**
- If `hoveredRef.current` is set, draw a tooltip near the cursor: dark background rect `#16213e` with 1px border `#2a2a4a`, text from `getTooltipText()`. Position: above the hovered building in screen space.

**Cleanup:**
- On unmount: `cancelAnimationFrame(animRef.current)`, disconnect ResizeObserver.

The canvas element should be `position: absolute, top: 0, left: 0, width: 100%, height: 100%` and set the actual width/height attributes via JS (not CSS).

**Note on dependencies:** The component will need to fetch asset_dependencies to draw dependency lines. Either pass them as a prop (from AssetsClient which includes them in the API call), or fetch separately. For simplicity, pass them as a prop. Update AssetsClient to also fetch `/api/assets` which should return dependency information. For now, skip dependency lines if the data isn't available — they can be wired when the detail panel loads full dependency data.
